    /**
     * Execute a SQL query against MongoDB collections
     * The collection to query is determined from the SQL statement (FROM clause)
     * @param sqlQuery The SQL query to execute
     * @return List of objects as maps with only the requested fields
     */
    public List<Map<String, Object>> executeQuery(String sqlQuery) {
        try {
            log.info("Executing SQL query: {}", sqlQuery);
            
            // Clean up the SQL query - remove any markdown formatting that might be present
            sqlQuery = cleanSqlQuery(sqlQuery);
            log.debug("Cleaned SQL query: {}", sqlQuery);
            
            Statement statement = CCJSqlParserUtil.parse(sqlQuery);
            
            if (statement instanceof Select) {
                Select select = (Select) statement;
                
                // Get the select body as PlainSelect
                if (!(select.getSelectBody() instanceof PlainSelect)) {
                    throw new IllegalArgumentException("Only simple SELECT statements are supported.");
                }
                
                // Using deprecated method but handling it properly
                @SuppressWarnings("deprecation")
                PlainSelect plainSelect = (PlainSelect) select.getSelectBody();
                
                // Extract table name from FROM clause
                if (plainSelect.getFromItem() == null) {
                    throw new IllegalArgumentException("FROM clause is required in the SQL query.");
                }
                
                if (!(plainSelect.getFromItem() instanceof Table)) {
                    throw new IllegalArgumentException("Only simple table names are supported in FROM clause.");
                }
                
                Table table = (Table) plainSelect.getFromItem();
                String tableName = table.getName().toLowerCase();
                
                // Map table name to collection name
                String collectionName = COLLECTION_MAPPING.getOrDefault(tableName, tableName);
                
                // Get the class for the collection
                Class<?> entityClass = CLASS_MAPPING.get(collectionName);
                if (entityClass == null) {
                    throw new IllegalArgumentException("Unknown collection: " + collectionName);
                }
                
                // Extract selected fields
                List<String> selectedFields = new ArrayList<>();
                boolean selectAll = false;
                
                for (SelectItem<?> item : plainSelect.getSelectItems()) {
                    String selectItemStr = item.toString();
                    log.debug("Processing select item: {}", selectItemStr);
                    
                    if (selectItemStr.equals("*")) {
                        selectAll = true;
                        break;
                    } else {
                        // Handle potential aliasing in the select item
                        if (selectItemStr.contains(" as ")) {
                            selectItemStr = selectItemStr.substring(0, selectItemStr.indexOf(" as ")).trim();
                        }
                        
                        // Remove quotes if present (for fields with dots)
                        if (selectItemStr.startsWith("\"") && selectItemStr.endsWith("\"")) {
                            selectItemStr = selectItemStr.substring(1, selectItemStr.length() - 1);
                        }
                        
                        log.debug("Adding field to selected fields: {}", selectItemStr);
                        selectedFields.add(selectItemStr);
                    }
                }
                
                log.debug("Selected fields: {}, selectAll: {}", selectedFields, selectAll);
                
                // Build MongoDB query
                Query query = new Query();
                
                // Add WHERE clause if present
                if (plainSelect.getWhere() != null) {
                    log.debug("WHERE clause present: {}", plainSelect.getWhere());
                    Criteria criteria = buildMongoCriteria(plainSelect.getWhere());
                    query.addCriteria(criteria);
                }
                
                log.debug("MongoDB query: {}", query);
                
                // Execute the query
                List<?> results = mongoTemplate.find(query, entityClass, collectionName);
                log.debug("Query returned {} results", results.size());
                
                // Convert results to maps with only the selected fields
                List<Map<String, Object>> mappedResults = new ArrayList<>();
                for (Object result : results) {
                    Map<String, Object> entityMap = convertEntityToMap(result, selectedFields, selectAll);
                    mappedResults.add(entityMap);
                }
                
                return mappedResults;
            } else {
                throw new IllegalArgumentException("Only SELECT statements are supported.");
            }
        } catch (Exception e) {
            log.error("Failed to parse or execute query: {}", e.getMessage(), e);
            throw new RuntimeException("Failed to parse or execute query: " + e.getMessage(), e);
        }
    }
    
    /**
     * Clean SQL query by removing any markdown formatting or other unwanted characters
     * @param sqlQuery The SQL query to clean
     * @return Cleaned SQL query
     */
    private String cleanSqlQuery(String sqlQuery) {
        // Remove markdown formatting
        sqlQuery = sqlQuery.replaceAll("\\[([^\\]]+)\\]\\(([^\\)]+)\\)", "$1");
    
        // Fix common errors from Postman / JSQLParser
        sqlQuery = sqlQuery.replaceAll("=\\s*\\\"(https?://.*?)\\\"", "= '$1'");
        sqlQuery = sqlQuery.replaceAll("=\\s*\\'(https?://.*?)\\s*$", "= '$1'");
    
        // Final cleanup: remove unbalanced quotes at the ends
        if (sqlQuery.startsWith("\"") && sqlQuery.endsWith("\"")) {
            sqlQuery = sqlQuery.substring(1, sqlQuery.length() - 1);
        }
    
        return sqlQuery;
    }

    /**
     * Extract nested field from an object using dot notation
     * @param field The field to extract (e.g., assetRef.type, submodelRefs.0.keys.0.value)
     * @param object The object to extract from
     * @param entityMap The map to store the extracted field
     */
    private void extractNestedFieldWithDotNotation(String field, Object object, Map<String, Object> entityMap) {
        try {
            String[] parts = normalizeIndexedPath(field).split("\\.");
            Object nestedObject = object;
            for (int i = 1; i < parts.length; i++) {
                String nestedField = parts[i];
                if (nestedObject == null) return;
                if (nestedField.matches("\\d+")) {
                    int index = Integer.parseInt(nestedField);
                    if (nestedObject instanceof List list && index < list.size()) nestedObject = list.get(index);
                    else return;
                } else {
                    try {
                        String getter = "get" + nestedField.substring(0, 1).toUpperCase() + nestedField.substring(1);
                        nestedObject = nestedObject.getClass().getMethod(getter).invoke(nestedObject);
                    } catch (Exception ex) {
                        try {
                            java.lang.reflect.Field f = nestedObject.getClass().getDeclaredField(nestedField);
                            f.setAccessible(true);
                            nestedObject = f.get(nestedObject);
                        } catch (Exception e) { return; }
                    }
                }
            }
            entityMap.put(field, nestedObject);
        } catch (Exception e) { }
    }

    /**
     * Handle nested field names in SQL queries
     * Converts SQL-style nested field notation to MongoDB dot notation
     * @param columnName The column name from SQL query
     * @return MongoDB compatible field name
     */
    private String handleNestedField(String columnName) {
        if (columnName.startsWith("\"") && columnName.endsWith("\"")) {
            columnName = columnName.substring(1, columnName.length() - 1);
        }
        return normalizeIndexedPath(columnName);
    }
    private String normalizeIndexedPath(String fieldPath) {
        // Inject .0 after any array-typed segment if not indexed already
        String[] parts = fieldPath.split("\\.");
        StringBuilder normalized = new StringBuilder();
        for (int i = 0; i < parts.length; i++) {
            if (i > 0) normalized.append(".");
            normalized.append(parts[i]);
            if (isArrayField(parts[i]) && (i + 1 >= parts.length || !parts[i + 1].matches("\\d+"))) {
                normalized.append(".0");
            }
        }
        return normalized.toString();
    }
    
    /**
     * Build MongoDB criteria from a JSQLParser expression
     * @param expression The JSQLParser expression
     * @return MongoDB criteria
     */
    private Criteria buildMongoCriteria(Expression expression) {
        if (expression instanceof GreaterThan) {
            GreaterThan greaterThan = (GreaterThan) expression;
            Column column = (Column) greaterThan.getLeftExpression();
            Expression rightExp = greaterThan.getRightExpression();
    
            String fieldName = handleNestedField(column.getColumnName());
            log.debug("Processing greater than condition for field: {}", fieldName);
    
            return buildNestedCriteria(fieldName, rightExp, "gt");
    
        } else if (expression instanceof EqualsTo) {
            EqualsTo equalsTo = (EqualsTo) expression;
            Column column = (Column) equalsTo.getLeftExpression();
            Expression rightExp = equalsTo.getRightExpression();
    
            String fieldName = handleNestedField(column.getColumnName());
            log.debug("Processing equals condition for field: {}", fieldName);
    
            return buildNestedCriteria(fieldName, rightExp, "eq");
    
        } else if (expression instanceof LikeExpression) {
            LikeExpression likeExp = (LikeExpression) expression;
            Column column = (Column) likeExp.getLeftExpression();
            Expression right = likeExp.getRightExpression();
    
            String fieldName = handleNestedField(column.getColumnName());
            String value = "";
    
            if (right instanceof StringValue) {
                value = ((StringValue) right).getValue().replace("%", ".*").replace("_", ".");
            } else {
                throw new RuntimeException("Unsupported LIKE pattern: " + right);
            }
    
            return buildNestedCriteria(fieldName, new StringValue(value), "regex");
    
        } else if (expression instanceof AndExpression) {
            AndExpression andExpression = (AndExpression) expression;
            Criteria leftCriteria = buildMongoCriteria(andExpression.getLeftExpression());
            Criteria rightCriteria = buildMongoCriteria(andExpression.getRightExpression());
            return new Criteria().andOperator(leftCriteria, rightCriteria);
    
        } else if (expression instanceof OrExpression) {
            OrExpression orExpression = (OrExpression) expression;
            Criteria leftCriteria = buildMongoCriteria(orExpression.getLeftExpression());
            Criteria rightCriteria = buildMongoCriteria(orExpression.getRightExpression());
            return new Criteria().orOperator(leftCriteria, rightCriteria);
        }
    
        throw new RuntimeException("Unsupported expression: " + expression);
    }
    private Criteria buildNestedCriteria(String path, Expression valueExpr, String operator) {
        String[] parts = path.split("\\.");
        Object value;
    
        if (valueExpr instanceof StringValue) {
            value = ((StringValue) valueExpr).getValue();
            if ("regex".equals(operator)) {
                value = value.toString().replace("%", ".*").replace("_", ".");
            }
        } else if (valueExpr instanceof LongValue) {
            value = ((LongValue) valueExpr).getValue();
        } else {
            throw new RuntimeException("Unsupported value expression: " + valueExpr);
        }
    
        return buildDeepCriteria(parts, 0, value, operator);
    }
    
    private Criteria buildDeepCriteria(String[] parts, int index, Object value, String operator) {
        String current = parts[index];
        boolean isLast = index == parts.length - 1;
    
        if (isLast) {
            return switch (operator) {
                case "eq" -> Criteria.where(current).is(value);
                case "gt" -> Criteria.where(current).gt(value);
                case "regex" -> Criteria.where(current).regex(value.toString(), "i");
                default -> throw new RuntimeException("Unsupported operator: " + operator);
            };
        }
    
        Criteria inner = buildDeepCriteria(parts, index + 1, value, operator);
    
        // Wrap with $elemMatch if next level is an array
        String nextPath = String.join(".", java.util.Arrays.copyOfRange(parts, 0, index + 2));
        if (isArrayField(nextPath)) {
            return Criteria.where(current).elemMatch(inner);
        } else {
            // Inline flatten path
            String flatPath = String.join(".", java.util.Arrays.copyOfRange(parts, 0, index + 2));
            return switch (operator) {
                case "eq" -> Criteria.where(flatPath).is(value);
                case "gt" -> Criteria.where(flatPath).gt(value);
                case "regex" -> Criteria.where(flatPath).regex(value.toString(), "i");
                default -> throw new RuntimeException("Unsupported operator: " + operator);
            };
        }
    }

    /**
 * Check if a field is an array field based on known array prefixes.
 * Helps decide whether to wrap the query with $elemMatch.
 *
 * @param fieldName The field name to check
 * @return true if the field is an array field or contains one in the path
 */
private boolean isArrayField(String fieldName) {
    return fieldName.matches("(?i).*\\b(submodelElements|qualifiers|semanticId\\.keys|supplementalSemanticIds|embeddedDataSpecifications|definition|preferredName|shortName)\\b.*");

}
